-- 1. ฟังก์ชันค้นหา Base Carpet ใน Plots
--------------------------------------------------------------------------------
local function findMyBaseCarpet()
	local plots = Workspace:FindFirstChild("Plots")
	if not plots then return nil end

	for _, baseFolder in ipairs(plots:GetChildren()) do
		local isMyBase = false
		
		local ownerVal = baseFolder:FindFirstChild("owner") or baseFolder:FindFirstChild("Owner")
		if ownerVal then
			if (ownerVal:IsA("StringValue") or ownerVal:IsA("TextLabel")) and ownerVal.Value == LocalPlayer.Name then
				isMyBase = true
			elseif ownerVal:IsA("ObjectValue") and ownerVal.Value == LocalPlayer then
				isMyBase = true
			end
		end

		if not isMyBase then
			for _, child in ipairs(baseFolder:GetDescendants()) do
				if child:IsA("StringValue") and child.Value == LocalPlayer.Name then
					isMyBase = true
					break
				end
			end
		end

		if isMyBase then
			local mainBase = baseFolder:FindFirstChild("MainBase")
			if mainBase then
				local carpet = mainBase:FindFirstChild("Carpet") or mainBase:FindFirstChild("GeneratedCarpets")
				if carpet then
					if carpet:IsA("BasePart") then
						return carpet
					elseif carpet:IsA("Model") then
						return carpet.PrimaryPart or carpet:FindFirstChildWhichIsA("BasePart")
					end
				end
			end
		end
	end

	for _, part in ipairs(plots:GetDescendants()) do
		if part.Name == "Carpet" and part:IsA("BasePart") then
			return part
		end
	end

	return nil
end

--------------------------------------------------------------------------------
-- 2. ฟังก์ชันค้นหา Guardian ที่ใกล้ตัวเราที่สุด
--------------------------------------------------------------------------------
local function getNearestGuardian()
	local guardiansFolder = Workspace:FindFirstChild("Live") and Workspace.Live:FindFirstChild("Guardians")
	local character = LocalPlayer.Character
	if not guardiansFolder or not character then return nil end

	local myHRP = character:FindFirstChild("HumanoidRootPart")
	if not myHRP then return nil end

	local nearestGuardian = nil
	local shortestDistance = math.huge

	for _, guardian in ipairs(guardiansFolder:GetChildren()) do
		if guardian:IsA("Model") then
			local gPart = guardian.PrimaryPart or guardian:FindFirstChild("HumanoidRootPart") or guardian:FindFirstChildWhichIsA("BasePart")
			if gPart then
				local dist = (myHRP.Position - gPart.Position).Magnitude
				if dist < shortestDistance then
					shortestDistance = dist
					nearestGuardian = guardian
				end
			end
		end
	end

	return nearestGuardian
end

--------------------------------------------------------------------------------
-- 3. ระบบขี่ Guardian และ Tween กลับบ้าน (ปรับยืนตัวตรง + ความเร็ว 1000)
--------------------------------------------------------------------------------
local function rideAndTweenGuardianToBase()
	local nearestGuardian = getNearestGuardian()
	local carpetPart = findMyBaseCarpet()
	local character = LocalPlayer.Character

	if not nearestGuardian or not carpetPart or not character then return end

	local myHRP = character:FindFirstChild("HumanoidRootPart")
	local gPart = nearestGuardian.PrimaryPart or nearestGuardian:FindFirstChild("HumanoidRootPart") or nearestGuardian:FindFirstChildWhichIsA("BasePart")

	if not myHRP or not gPart then return end

	-- ปรับตำแหน่งการวางให้ตัว Guardian และ ตัวเราตั้งตรง ไม่เอียงตามองศาเดิม
	local currentGPos = gPart.Position
	local uprightGCFrame = CFrame.lookAt(currentGPos, currentGPos + gPart.CFrame.LookVector * Vector3.new(1, 0, 1), Vector3.new(0, 1, 0))
	nearestGuardian:PivotTo(uprightGCFrame)

	-- ย้ายตัวเราขึ้นไปบนหลัง Guardian ในท่าตั้งตรง
	character:PivotTo(uprightGCFrame * CFrame.new(0, 3, 1))

	local weld = Instance.new("WeldConstraint")
	weld.Part0 = myHRP
	weld.Part1 = gPart
	weld.Parent = myHRP

	-- กำหนด CFrame เป้าหมายโดยบังคับให้แกนตั้งตรงกับพื้นโลกเสมอ
	local targetPos = carpetPart.Position + Vector3.new(0, 3, 0)
	local targetCFrame = CFrame.lookAt(targetPos, targetPos + uprightGCFrame.LookVector, Vector3.new(0, 1, 0))

	local distance = (gPart.Position - targetPos).Magnitude
	local speed = 1000
	local tweenTime = math.max(distance / speed, 0.05)

	if not nearestGuardian.PrimaryPart then
		nearestGuardian.PrimaryPart = gPart
	end

	local tweenInfo = TweenInfo.new(tweenTime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
	local cframeValue = Instance.new("CFrameValue")
	cframeValue.Value = nearestGuardian:GetPivot()

	local tween = TweenService:Create(cframeValue, tweenInfo, {Value = targetCFrame})

	local connection
	connection = cframeValue.Changed:Connect(function(newCFrame)
		nearestGuardian:PivotTo(newCFrame)
	end)

	tween:Play()
	tween.Completed:Wait()

	connection:Disconnect()
	cframeValue:Destroy()

	-- ถอด Weld และตั้งตัวเราให้ยืนตรงบน Carpet
	task.wait(0.1)
	weld:Destroy()
	character:PivotTo(CFrame.lookAt(targetPos, targetPos + character.HumanoidRootPart.CFrame.LookVector * Vector3.new(1, 0, 1), Vector3.new(0, 1, 0)))
end

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragging = false end
		end)

local selectedModelName = nil
local currentViewportModel = nil
local listViewportModels = {}

local function setViewportModel(model)
	task.spawn(function()
		if currentViewportModel then currentViewportModel:Destroy() currentViewportModel = nil end
		if not model then return end

		local clone = model:Clone()
		pcall(function() ContentProvider:PreloadAsync({clone}) end)

		clone.Parent = ViewportFrame
		currentViewportModel = clone

		local primary = clone.PrimaryPart or clone:FindFirstChildWhichIsA("BasePart")
		if primary then
			local cframe, size = clone:GetBoundingBox()
			local maxDim = math.max(size.X, size.Y, size.Z)
			ViewportCamera.CFrame = CFrame.new(cframe.Position + Vector3.new(0, maxDim * 0.25, maxDim * 1.7), cframe.Position)
		end
	end)
end

RunService.RenderStepped:Connect(function(dt)
	if MainFrame.Visible then
		if currentViewportModel then
			local cframe, _ = currentViewportModel:GetBoundingBox()
			currentViewportModel:PivotTo(cframe * CFrame.Angles(0, math.rad(40 * dt), 0))
		end

		if DropdownList.Visible then
			for _, itemData in ipairs(listViewportModels) do
				if itemData.model then
					local cframe, _ = itemData.model:GetBoundingBox()
					itemData.model:PivotTo(cframe * CFrame.Angles(0, math.rad(40 * dt), 0))
				end
			end
		end
	end
end)

-- 8. Integrated Main Steal & Ride Loop
--------------------------------------------------------------------------------
local isSystemActive = false

local function triggerPrompt(prompt)
	if fireproximityprompt then
		fireproximityprompt(prompt)
	elseif prompt:FindFirstChild("InputHoldBegan") then
		prompt:InputHoldBegan()
		task.wait(prompt.HoldDuration or 0.1)
		prompt:InputHoldEnd()
	end
end

SwitchBtn.MouseButton1Click:Connect(function()
	isSystemActive = not isSystemActive

	if isSystemActive then
		TweenService:Create(SwitchBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(16, 185, 129)}):Play()
		TweenService:Create(SwitchKnob, TweenInfo.new(0.2), {Position = UDim2.new(1, -23, 0.5, -10), BackgroundColor3 = Color3.fromRGB(255, 255, 255)}):Play()

		task.spawn(function()
			while isSystemActive do
				if selectedModelName then
					local slimesFolder = Workspace:FindFirstChild("Live") and Workspace.Live:FindFirstChild("Slimes")
					if slimesFolder then
						local targetSlime = slimesFolder:FindFirstChild(selectedModelName)
						local character = LocalPlayer.Character
						if targetSlime and character then
							local targetPart = targetSlime.PrimaryPart or targetSlime:FindFirstChildWhichIsA("BasePart")
							if targetPart then
								-- Step 1: วาร์ปตัวเราไปเก็บ Slime โดยปรับให้ตัวเรายืนตรงเสมอ
								local currentStealPos = targetPart.Position + Vector3.new(0, 2, 3)
								character:PivotTo(CFrame.lookAt(currentStealPos, targetPart.Position, Vector3.new(0, 1, 0)))
								task.wait(0.25)

								-- Step 2: กด ProximityPrompt Steal
								local rootPart = targetSlime:FindFirstChild("RootPart")
								if rootPart then
									local prompt = rootPart:FindFirstChild("StealPrompt") or rootPart:FindFirstChildWhichIsA("ProximityPrompt", true)
									if prompt then triggerPrompt(prompt) end
								end
