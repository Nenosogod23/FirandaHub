-- 1. ฟังก์ชันค้นหาจุด Start (ปลายทางสำหรับ Tween กลับ)
--------------------------------------------------------------------------------
local function getStartTarget()
	local mapFolder = Workspace:FindFirstChild("Map")
	if not mapFolder then return nil end

	local startObj = mapFolder:FindFirstChild("Start")
	if not startObj then return nil end

	if startObj:IsA("BasePart") then
		return startObj
	elseif startObj:IsA("Model") then
		return startObj.PrimaryPart or startObj:FindFirstChildWhichIsA("BasePart")
	end

	return nil
end

--------------------------------------------------------------------------------
-- 2. ฟังก์ชันค้นหา Guardian ที่ใกล้ตัวเราที่สุดใน Workspace.Map.ClientGuards
--------------------------------------------------------------------------------
local function getNearestGuardian()
	local mapFolder = Workspace:FindFirstChild("Map")
	local guardsFolder = mapFolder and mapFolder:FindFirstChild("ClientGuards") or Workspace:FindFirstChild("ClientGuards")
	local character = LocalPlayer.Character
	if not guardsFolder or not character then return nil end

	local myHRP = character:FindFirstChild("HumanoidRootPart")
	if not myHRP then return nil end

	local nearestGuardian = nil
	local shortestDistance = math.huge

	for _, guardian in ipairs(guardsFolder:GetChildren()) do
		if guardian:IsA("Model") or guardian:IsA("BasePart") then
			local gPart = guardian:IsA("Model") and (guardian.PrimaryPart or guardian:FindFirstChild("HumanoidRootPart") or guardian:FindFirstChildWhichIsA("BasePart")) or guardian
			if gPart then
				local dist = (myHRP.Position - gPart.Position).Magnitude
				if dist < shortestDistance then
					shortestDistance = dist
					nearestGuardian = guardian
				end
			end
		end
	end

	return nearestGuardian
end

--------------------------------------------------------------------------------
-- 3. ระบบขี่ Guardian และ Tween ไปยัง Workspace.Map.Start (ความเร็ว 1000)
--------------------------------------------------------------------------------
local function rideAndTweenGuardianToStart()
	local nearestGuardian = getNearestGuardian()
	local startPart = getStartTarget()
	local character = LocalPlayer.Character

	if not nearestGuardian or not startPart or not character then return end

	local myHRP = character:FindFirstChild("HumanoidRootPart")
	local gPart = nearestGuardian:IsA("Model") and (nearestGuardian.PrimaryPart or nearestGuardian:FindFirstChild("HumanoidRootPart") or nearestGuardian:FindFirstChildWhichIsA("BasePart")) or nearestGuardian

	if not myHRP or not gPart then return end

	-- ปรับตำแหน่งการวางให้ตัว Guardian และ ตัวเราตั้งตรง ไม่เอียง
	local currentGPos = gPart.Position
	local uprightGCFrame = CFrame.lookAt(currentGPos, currentGPos + gPart.CFrame.LookVector * Vector3.new(1, 0, 1), Vector3.new(0, 1, 0))
	
	if nearestGuardian:IsA("Model") then
		nearestGuardian:PivotTo(uprightGCFrame)
	else
		gPart.CFrame = uprightGCFrame
	end

	-- ย้ายตัวเราขึ้นไปบนหลัง Guardian ในท่าตั้งตรง
	character:PivotTo(uprightGCFrame * CFrame.new(0, 3, 1))

	local weld = Instance.new("WeldConstraint")
	weld.Part0 = myHRP
	weld.Part1 = gPart
	weld.Parent = myHRP

	-- กำหนด CFrame เป้าหมายไปที่ Workspace.Map.Start
	local targetPos = startPart.Position + Vector3.new(0, 3, 0)
	local targetCFrame = CFrame.lookAt(targetPos, targetPos + uprightGCFrame.LookVector, Vector3.new(0, 1, 0))

	local distance = (gPart.Position - targetPos).Magnitude
	local speed = 1000
	local tweenTime = math.max(distance / speed, 0.05)

	if nearestGuardian:IsA("Model") and not nearestGuardian.PrimaryPart then
		nearestGuardian.PrimaryPart = gPart
	end

	local tweenInfo = TweenInfo.new(tweenTime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
	local cframeValue = Instance.new("CFrameValue")
	cframeValue.Value = nearestGuardian:IsA("Model") and nearestGuardian:GetPivot() or gPart.CFrame

	local tween = TweenService:Create(cframeValue, tweenInfo, {Value = targetCFrame})

	local connection
	connection = cframeValue.Changed:Connect(function(newCFrame)
		if nearestGuardian:IsA("Model") then
			nearestGuardian:PivotTo(newCFrame)
		else
			gPart.CFrame = newCFrame
		end
	end)

	tween:Play()
	tween.Completed:Wait()

	connection:Disconnect()
	cframeValue:Destroy()

	-- ถอด Weld และตั้งตัวเราให้ยืนตรงบนจุด Start
	task.wait(0.1)
	weld:Destroy()
	character:PivotTo(CFrame.lookAt(targetPos, targetPos + character.HumanoidRootPart.CFrame.LookVector * Vector3.new(1, 0, 1), Vector3.new(0, 1, 0)))
end

local function setViewportModel(model)
	task.spawn(function()
		if currentViewportModel then currentViewportModel:Destroy() currentViewportModel = nil end
		if not model then return end

		local clone = model:Clone()
		pcall(function() ContentProvider:PreloadAsync({clone}) end)

		clone.Parent = ViewportFrame
		currentViewportModel = clone

		local primary = clone:IsA("Model") and (clone.PrimaryPart or clone:FindFirstChildWhichIsA("BasePart")) or clone
		if primary then
			local cframe, size
			if clone:IsA("Model") then
				cframe, size = clone:GetBoundingBox()
			else
				cframe = clone.CFrame
				size = clone.Size
			end
			local maxDim = math.max(size.X, size.Y, size.Z)
			ViewportCamera.CFrame = CFrame.new(cframe.Position + Vector3.new(0, maxDim * 0.25, maxDim * 1.7), cframe.Position)
		end
	end)
end

			end

	end
end)

local isSystemActive = false

local function triggerPrompt(prompt)
	if fireproximityprompt then
		fireproximityprompt(prompt)
	elseif prompt:FindFirstChild("InputHoldBegan") then
		prompt:InputHoldBegan()
		task.wait(prompt.HoldDuration or 0.1)
		prompt:InputHoldEnd()
	end
end

SwitchBtn.MouseButton1Click:Connect(function()
	isSystemActive = not isSystemActive

	if isSystemActive then
		TweenService:Create(SwitchBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(16, 185, 129)}):Play()
		TweenService:Create(SwitchKnob, TweenInfo.new(0.2), {Position = UDim2.new(1, -23, 0.5, -10), BackgroundColor3 = Color3.fromRGB(255, 255, 255)}):Play()

		task.spawn(function()
			while isSystemActive do
				if selectedTargetObj and selectedTargetObj.Parent then
					local character = LocalPlayer.Character
					if character then
						local targetPart = selectedTargetObj:IsA("Model") and (selectedTargetObj.PrimaryPart or selectedTargetObj:FindFirstChildWhichIsA("BasePart")) or selectedTargetObj
						if targetPart then
							-- Step 1: วาร์ปตัวเราไปหา Egg ที่เลือก ยืนตรง
							local currentStealPos = targetPart.Position + Vector3.new(0, 2, 3)
							character:PivotTo(CFrame.lookAt(currentStealPos, targetPart.Position, Vector3.new(0, 1, 0)))
							task.wait(0.25)

							-- Step 2: กด ProximityPrompt Steal
							local prompt = selectedTargetObj:FindFirstChildWhichIsA("ProximityPrompt", true) or targetPart:FindFirstChildWhichIsA("ProximityPrompt", true)
							if prompt then triggerPrompt(prompt) end

							task.wait(0.25)

							-- Step 3: ขี่ Guardian ยืนตัวตรง Tween ไป Workspace.Map.Start ความเร็ว 1000
							rideAndTweenGuardianToStart()
						end
					end
				end
				task.wait(1.2)
			end
