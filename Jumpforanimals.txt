-- Helper Function: Auto Focus Camera
--------------------------------------------------------------------------------
local function setupViewportCamera(viewport, camera, model)
	if not model then return end
	
	for _, oldChild in ipairs(viewport:GetChildren()) do
		if oldChild:IsA("Model") or oldChild:IsA("BasePart") then
			oldChild:Destroy()
		end
	end

	local clone = model:Clone()
	clone.Parent = viewport
	
	for _, desc in ipairs(clone:GetDescendants()) do
		if desc:IsA("LuaSourceContainer") then
			desc.Disabled = true
		end
	end

	local cf, size
	if clone:IsA("Model") then
		cf, size = clone:GetBoundingBox()
		if size.Magnitude == 0 then
			local primary = clone.PrimaryPart or clone:FindFirstChildWhichIsA("BasePart", true)
			if primary then
				cf = primary.CFrame
				size = primary.Size
			end
		end
	elseif clone:IsA("BasePart") then
		cf = clone.CFrame
		size = clone.Size
	end

	if not cf or not size then return clone end

	local maxDim = math.max(size.X, size.Y, size.Z)
	if maxDim == 0 then maxDim = 2 end
	
	local fovRad = math.rad(camera.FieldOfView / 2)
	local distance = (maxDim / (2 * math.tan(fovRad))) * 1.45

	local offset = Vector3.new(distance * 0.4, distance * 0.3, distance * 0.9)
	camera.CFrame = CFrame.lookAt(cf.Position + offset, cf.Position)

	return clone
end

-- Helper Function: Trigger Prompt (Instant Hold = 0)
--------------------------------------------------------------------------------
local function interactWithEgg(eggObj)
	if not eggObj then return end

	-- 1. ปรับ ProximityPrompt ให้ HoldDuration = 0 แล้วกดทันที
	for _, prompt in ipairs(eggObj:GetDescendants()) do
		if prompt:IsA("ProximityPrompt") then
			prompt.HoldDuration = 0
			if fireproximityprompt then
				fireproximityprompt(prompt)
			else
				prompt:InputHoldBegin()
				task.wait(0.02)
				prompt:InputHoldEnd()
			end
		end
	end

	-- 2. กดปุ่ม E สำรอง 1 ครั้ง
	VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
	task.wait(0.05)
	VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
end

-- 1. ฟังก์ชัน Auto Farm: ค้นหา Plot และ Teleport ไปยัง SquatZone.Floor
--------------------------------------------------------------------------------
local function findMyPlotAndTeleport()
	local mapFolder = Workspace:FindFirstChild("Map")
	local plotsFolder = mapFolder and mapFolder:FindFirstChild("Plots") or Workspace:FindFirstChild("Plots")
	if not plotsFolder then return false end

	local myName = LocalPlayer.Name
	local myDisplayName = LocalPlayer.DisplayName

	for _, plot in ipairs(plotsFolder:GetChildren()) do
		for _, child in ipairs(plot:GetDescendants()) do
			if child:IsA("StringValue") then
				if child.Value == myName or child.Value == myDisplayName then
					local squatZone = plot:FindFirstChild("SquatZone", true)
					if squatZone then
						local targetFloor = squatZone:FindFirstChild("Floor", true) or squatZone
						local character = LocalPlayer.Character
						if character and character:FindFirstChild("HumanoidRootPart") then
							local targetCFrame = targetFloor:IsA("BasePart") and targetFloor.CFrame or targetFloor:GetPivot()
							character:PivotTo(targetCFrame + Vector3.new(0, 3, 0))
							return true
						end
					end
				end
			end
		end
	end
	return false
end

--------------------------------------------------------------------------------
-- 2. ฟังก์ชันขี่ Guardian (ถอยระยะ 60 Studs & แช่แข็งตัวละคร)
--------------------------------------------------------------------------------
local function getStartTarget()
	local mapFolder = Workspace:FindFirstChild("Map")
	if not mapFolder then return nil end
	local startObj = mapFolder:FindFirstChild("Start")
	if not startObj then return nil end
	return startObj:IsA("BasePart") and startObj or (startObj:IsA("Model") and (startObj.PrimaryPart or startObj:FindFirstChildWhichIsA("BasePart")))
end

local function getNearestGuardian()
	local mapFolder = Workspace:FindFirstChild("Map")
	local guardsFolder = mapFolder and mapFolder:FindFirstChild("ClientGuards") or Workspace:FindFirstChild("ClientGuards")
	local character = LocalPlayer.Character
	if not guardsFolder or not character or not character:FindFirstChild("HumanoidRootPart") then return nil end

	local myHRP = character.HumanoidRootPart
	local nearestGuardian, shortestDistance = nil, math.huge

	for _, guardian in ipairs(guardsFolder:GetChildren()) do
		local gPart = guardian:IsA("Model") and (guardian.PrimaryPart or guardian:FindFirstChild("HumanoidRootPart") or guardian:FindFirstChildWhichIsA("BasePart")) or guardian
		if gPart then
			local dist = (myHRP.Position - gPart.Position).Magnitude
			if dist < shortestDistance then
				shortestDistance = dist
				nearestGuardian = guardian
			end
		end
	end
	return nearestGuardian
end

local function rideAndTweenGuardianToStart()
	local nearestGuardian = getNearestGuardian()
	local startPart = getStartTarget()
	local character = LocalPlayer.Character
	if not nearestGuardian or not startPart or not character then return end

	local myHRP = character:FindFirstChild("HumanoidRootPart")
	local myHum = character:FindFirstChildOfClass("Humanoid")
	local gPart = nearestGuardian:IsA("Model") and (nearestGuardian.PrimaryPart or nearestGuardian:FindFirstChild("HumanoidRootPart") or nearestGuardian:FindFirstChildWhichIsA("BasePart")) or nearestGuardian
	local gHum = nearestGuardian:FindFirstChildOfClass("Humanoid")
	
	if not myHRP or not gPart then return end

	-- แช่แข็ง Guardian (ยืนตัวตรง ไม่ขยับ)
	if gHum then
		gHum.PlatformStand = true
		gHum.WalkSpeed = 0
		gHum.JumpPower = 0
	end

	-- แช่แข็งผู้เล่น
	if myHum then
		myHum.PlatformStand = true
	end

	-- จัดแนว Guardian ให้ยืนตรงตั้งฉากกับพื้น
	local currentGPos = gPart.Position
	local uprightGCFrame = CFrame.lookAt(currentGPos, currentGPos + gPart.CFrame.LookVector * Vector3.new(1, 0, 1), Vector3.new(0, 1, 0))
	
	if nearestGuardian:IsA("Model") then nearestGuardian:PivotTo(uprightGCFrame) else gPart.CFrame = uprightGCFrame end
	
	-- ถอยระยะผู้เล่นออกไปด้านหลัง 60 Studs
	local playerOffset = CFrame.new(0, 4, 60)
	character:PivotTo(uprightGCFrame * playerOffset)

	local weld = Instance.new("WeldConstraint")
	weld.Part0 = myHRP
	weld.Part1 = gPart
	weld.Parent = myHRP

	local targetPos = startPart.Position + Vector3.new(0, 3, 0)
	local targetCFrame = CFrame.lookAt(targetPos, targetPos + uprightGCFrame.LookVector, Vector3.new(0, 1, 0))
	local tweenTime = math.max((gPart.Position - targetPos).Magnitude / Config.TweenSpeed, 0.05)

	if nearestGuardian:IsA("Model") and not nearestGuardian.PrimaryPart then nearestGuardian.PrimaryPart = gPart end

	local cframeValue = Instance.new("CFrameValue")
	cframeValue.Value = nearestGuardian:IsA("Model") and nearestGuardian:GetPivot() or gPart.CFrame

	local tween = TweenService:Create(cframeValue, TweenInfo.new(tweenTime, Enum.EasingStyle.Linear), {Value = targetCFrame})
	local connection = cframeValue.Changed:Connect(function(newCFrame)
		if nearestGuardian:IsA("Model") then nearestGuardian:PivotTo(newCFrame) else gPart.CFrame = newCFrame end
	end)

	tween:Play()
	tween.Completed:Wait()

	connection:Disconnect()
	cframeValue:Destroy()
	task.wait(0.1)
	
	-- ปลดล็อคและคืนค่าผู้เล่น
	weld:Destroy()
	if myHum then
		myHum.PlatformStand = false
	end
	
	character:PivotTo(CFrame.lookAt(targetPos, targetPos + myHRP.CFrame.LookVector * Vector3.new(1, 0, 1), Vector3.new(0, 1, 0)))
end

		if mouse and character and character:FindFirstChild("HumanoidRootPart") then
			character:PivotTo(CFrame.new(mouse.Hit.Position + Vector3.new(0, 3, 0)))
		end
	end)
	tpTool.Parent = backpack
end

local function refreshEggsFromStages(filterText)
	filterText = filterText and filterText:lower() or ""
	
	for _, child in ipairs(DropdownList:GetChildren()) do 
		if child:IsA("TextButton") then child:Destroy() end 
	end

	local mapFolder = Workspace:FindFirstChild("Map")
	local stagesFolder = mapFolder and mapFolder:FindFirstChild("Stages") or Workspace:FindFirstChild("Stages")
	if not stagesFolder then return end

	local seenEggNames = {}
	local count = 0

	for _, stage in ipairs(stagesFolder:GetChildren()) do
		local spawnedEggsFolder = stage:FindFirstChild("SpawnedEggs")
		if spawnedEggsFolder then
			for _, egg in ipairs(spawnedEggsFolder:GetChildren()) do
				local fullTitle = stage.Name .. " - " .. egg.Name
				
				if not seenEggNames[fullTitle] then
					if filterText == "" or fullTitle:lower():find(filterText, 1, true) then
						seenEggNames[fullTitle] = true
						count = count + 1
						
						local itemBtn = Instance.new("TextButton")
						itemBtn.Size = UDim2.new(1, -12, 0, 34)
						itemBtn.BackgroundColor3 = Color3.fromRGB(28, 32, 45)
						itemBtn.BackgroundTransparency = 0.3
						itemBtn.Text = ""
						itemBtn.ZIndex = 22
						itemBtn.Parent = DropdownList
						
						local itemCorner = Instance.new("UICorner") 
						itemCorner.CornerRadius = UDim.new(0, 6) 
						itemCorner.Parent = itemBtn

						local miniViewportBg = Instance.new("Frame")
						miniViewportBg.Size = UDim2.new(0, 26, 0, 26)
						miniViewportBg.Position = UDim2.new(0, 4, 0.5, -13)
						miniViewportBg.BackgroundColor3 = Color3.fromRGB(16, 18, 25)
						miniViewportBg.ZIndex = 23
						miniViewportBg.Parent = itemBtn
						local miniBgCorner = Instance.new("UICorner") miniBgCorner.CornerRadius = UDim.new(0, 4) miniBgCorner.Parent = miniViewportBg

						local miniViewport = Instance.new("ViewportFrame")
						miniViewport.Size = UDim2.new(1, 0, 1, 0)
						miniViewport.BackgroundTransparency = 1
						miniViewport.ZIndex = 24
						miniViewport.Parent = miniViewportBg

						local miniCamera = Instance.new("Camera")
						miniCamera.FieldOfView = 50
						miniViewport.CurrentCamera = miniCamera
						miniCamera.Parent = miniViewport

						setupViewportCamera(miniViewport, miniCamera, egg)

						local itemLabel = Instance.new("TextLabel")
						itemLabel.Text = fullTitle
						itemLabel.Font = Enum.Font.GothamMedium
						itemLabel.TextColor3 = Color3.fromRGB(210, 220, 240)
						itemLabel.TextSize = 11
						itemLabel.TextXAlignment = Enum.TextXAlignment.Left
						itemLabel.Position = UDim2.new(0, 36, 0, 0)
						itemLabel.Size = UDim2.new(1, -40, 1, 0)
						itemLabel.BackgroundTransparency = 1
						itemLabel.ZIndex = 23
						itemLabel.Parent = itemBtn

						itemBtn.MouseButton1Click:Connect(function()
							selectedTargetName = egg.Name
							SelectedText.Text = egg.Name
							setViewportModel(egg)
							DropdownContainer.Visible = false
							Arrow.Text = "▼"
						end)
					end
				end
			end
		end
	end
	DropdownList.CanvasSize = UDim2.new(0, 0, 0, count * 38)
end

--------------------------------------------------------------------------------
local function findEggByName(name)
	local mapFolder = Workspace:FindFirstChild("Map")
	local stagesFolder = mapFolder and mapFolder:FindFirstChild("Stages") or Workspace:FindFirstChild("Stages")
	if not stagesFolder then return nil end

	for _, stage in ipairs(stagesFolder:GetChildren()) do
		local spawnedEggsFolder = stage:FindFirstChild("SpawnedEggs")
		if spawnedEggsFolder then
			local egg = spawnedEggsFolder:FindFirstChild(name)
			if egg then return egg end
		end
	end
	return nil
end

		TweenService:Create(SwitchBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(16, 185, 129)}):Play()

				if selectedTargetName then
					local targetEgg = findEggByName(selectedTargetName)
					if targetEgg and targetEgg.Parent then
						local character = LocalPlayer.Character
						if character then
							local targetPart = targetEgg:IsA("Model") and (targetEgg.PrimaryPart or targetEgg:FindFirstChildWhichIsA("BasePart")) or targetEgg

								-- Teleport ไปยังไข่

								task.wait(0.2)

		end)
	else
		TweenService:Create(SwitchBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(35, 40, 52)}):Play()
		TweenService:Create(SwitchKnob, TweenInfo.new(0.2), {Position = UDim2.new(0, 3, 0.5, -10), BackgroundColor3 = Color3.fromRGB(180, 190, 210)}):Play()
